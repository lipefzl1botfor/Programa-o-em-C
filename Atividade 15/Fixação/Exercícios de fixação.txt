2. Armazenar as notas de três alunos em duas provas e calcular a média de cada aluno
#include <stdio.h>

int main()
{
    float nota[3][2];
    float media[3]; 

    printf("Por favor digite as notas de cada aluno:\n");

    for(int i = 0; i < 3; i++){
        float soma_notas = 0; 
        
        for(int j = 0; j < 2; j++){
            printf("Por favor digite valor para posicao [%d] [%d]: ", i, j);
            scanf("%f", &nota[i][j]);
            
            soma_notas = soma_notas + nota[i][j]; 
        }
        
       
        media[i] = soma_notas / 2.0; 
    }

    printf("\n--- Seus Valores e Medias ---\n");

    for(int i = 0; i < 3; i++){
        printf("Aluno %d:\t", i + 1);
        
        for(int j = 0; j < 2; j++){
            printf("%.1f\t", nota[i][j]); 
        }
        
        printf("-> Media: %.1f\n", media[i]); 
    }

    return 0;
}

3. Criar uma matriz 3×3 e contar quantos valores são pares.
#include <stdio.h>

int main()
{
    int matriz[3][3];
    int contador_pares = 0; 

    printf("Por favor, digite os valores para a matriz 3x3:\n");
    
    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            printf("Digite o valor para a posicao [%d][%d]: ", i, j);
            scanf("%d", &matriz[i][j]);
            if(matriz[i][j] % 2 == 0) {
                contador_pares++; 
            }
        }
    }

    printf("\n--- Sua Matriz ---\n");

    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            printf("%d\t", matriz[i][j]);
        }
        printf("\n");
    }

    printf("\nA quantidade de valores pares na matriz e: %d\n", contador_pares);

    return 0;
}

4. Criar uma matriz 3×3 e identificar o maior valor armazenado.
#include <stdio.h>

int main()
{
    int matriz[3][3];
    int maior; 

    printf("Por favor, digite os valores para a matriz 3x3:\n");
    
    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            printf("Digite o valor para a posicao [%d][%d]: ", i, j);
            scanf("%d", &matriz[i][j]);
        }
    }

    printf("\n--- Sua Matriz ---\n");

    maior= matriz [0][0];
    
    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            printf("%d\t", matriz[i][j]);
            if(matriz [i][j] > maior){
                maior = matriz [i][j];
            }
        }
        printf("\n");
    }

    printf("\nO maior valor na matriz é: %d\n", maior);

    return 0;
}
